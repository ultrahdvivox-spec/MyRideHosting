const destination = $("destination").value.trim(); const vehicle = normalizeVehicleType(selectedVehicle()); const fare = Number(routeFare); if (!name) { $("bookMsg").textContent = "अपना पूरा नाम भरें।"; return; } if (!/^\d{10}$/.test(phone)) { $("bookMsg").textContent = "सही 10 अंकों का मोबाइल नंबर भरें।"; return; } if (alternatePhone && !/^\d{10}$/.test(alternatePhone)) { $("bookMsg").textContent = "Alternate Number सही भरें या खाली छोड़ें।"; return; } if (!pickup || !destination) { $("bookMsg").textContent = "Pickup और Destination भरें।"; return; } if (pickup.toLowerCase() === destination.toLowerCase()) { $("bookMsg").textContent = "दोनों जगह अलग-अलग रखें।"; return; } if (!vehicle) { $("bookMsg").textContent = "कृपया सही वाहन चुनें।"; return; } if (routeFare === null || routeDistanceKm === null || !Number.isFinite(fare) || fare <= 0 || fare > 100000) { $("bookMsg").textContent = "पहले सही Route और किराया Calculate करें।"; if (alternatePhone) { bookingData.alternatePhone = alternatePhone; } const bookingRef = await addDoc(collection(db, "Booking"), bookingData); $("bookMsg").textContent = "✅ Booking सफल!
