// ====================================== // HOTEL MAP + FREE OPENSTREETMAP SEARCH // ====================================== let hotelMap=null; let hotelMarker=null; let hotelSearchTimer=null; let hotelSearchController=null; let selectedHotelLocation=false; function initHotelMap(){ if(hotelMap)return; if(typeof L==="undefined"){ say("hotelMapMsg","Map Library लोड नहीं हुई। Internet जाँचें।"); return; } hotelMap=L.map("hotelMap").setView([25.34,74.63],12); L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png",{ maxZoom:19, attribution:"© OpenStreetMap contributors" }).addTo(hotelMap); // Map पर टैप करके Location चुनें hotelMap.on("click",async e=>{ await setHotelLocation(e.latlng.lat,e.latlng.lng,true); }); } function makeHotelAddress(p){ return [ p.name, p.housenumber, p.street, p.locality, p.district, p.city, p.state, p.postcode, p.country ].filter((v,i,a)=>v&&a.indexOf(v)===i).join(", "); } function showHotelOnMap(lat,lng){ initHotelMap(); if(!hotelMap)return; if(hotelMarker){ hotelMap.removeLayer(hotelMarker); } hotelMarker=L.marker([lat,lng],{ draggable:true }).addTo(hotelMap); hotelMarker.bindPopup( "Hotel Location
Pin को खिसकाकर सही जगह चुनें" ).openPopup(); hotelMap.setView([lat,lng],18); // Pin खिसकाने पर GPS बदलें hotelMarker.on("dragend",async()=>{ const pos=hotelMarker.getLatLng(); await setHotelLocation(pos.lat,pos.lng,true,false); }); setTimeout(()=>hotelMap.invalidateSize(),300); } async function reverseHotelAddress(lat,lng){ try{ const url= "https://photon.komoot.io/reverse?lat="+ encodeURIComponent(lat)+ "&lon="+encodeURIComponent(lng)+ "&lang=en"; const response=await fetch(url); if(!response.ok)return ""; const data=await response.json(); const feature=data.features?.[0]; if(!feature)return ""; return makeHotelAddress(feature.properties||""); }catch(e){ return ""; } } async function setHotelLocation(lat,lng,reverse=true,moveMarker=true){ lat=Number(lat); lng=Number(lng); if(!Number.isFinite(lat)||Math.abs(lat)>90|| !Number.isFinite(lng)||Math.abs(lng)>180){ say("hotelMapMsg","GPS Location सही नहीं है।"); return; } $("hotelLat").value=lat.toFixed(7); $("hotelLng").value=lng.toFixed(7); selectedHotelLocation=true; if(moveMarker){ showHotelOnMap(lat,lng); } const mapsUrl= "https://www.google.com/maps/search/?api=1&query="+ encodeURIComponent(lat+","+lng); $("hotelGoogleMapsLink").href=mapsUrl; $("hotelGoogleMapsLink").classList.remove("hidden"); say("hotelMapMsg","GPS Location चुनी गई। Address जाँचा जा रहा है..."); if(reverse){ const address=await reverseHotelAddress(lat,lng); if(address){ $("hotelAddress").value=address; } say( "hotelMapMsg", "Location चुनी गई। Latitude: "+lat.toFixed(7)+ " | Longitude: "+lng.toFixed(7)+ (address?" | Address अपडेट हो गया।":" | Address हाथ से जाँचें।") ); }else{ say("hotelMapMsg","GPS Location अपडेट हो गई।"); } } function clearHotelLocation(){ $("hotelAddress").value=""; $("hotelLat").value=""; $("hotelLng").value=""; $("hotelGoogleMapsLink").classList.add("hidden"); $("hotelGoogleMapsLink").removeAttribute("href"); if(hotelMarker&&hotelMap){ hotelMap.removeLayer(hotelMarker); hotelMarker=null; } selectedHotelLocation=false; say("hotelMapMsg","Location Clear हो गई।"); } function selectHotelResult(feature){ const p=feature.properties||{}; const coords=feature.geometry?.coordinates; if(!coords||coords.length<2){ say("hotelSearchMsg","इस परिणाम की GPS Location उपलब्ध नहीं है।"); return; } const lng=Number(coords[0]); const lat=Number(coords[1]); if(!Number.isFinite(lat)||!Number.isFinite(lng)){ say("hotelSearchMsg","GPS Coordinates सही नहीं हैं।"); return; } const name=p.name||$("hotelName").value.trim(); const address=makeHotelAddress(p); $("hotelName").value=name; $("hotelAddress").value=address||name; if(p.city){ $("hotelCity").value=p.city; }else if(p.district){ $("hotelCity").value=p.district; } $("hotelSearchResults").replaceChildren(); setHotelLocation(lat,lng,false,true); say("hotelSearchMsg","Hotel Location चुन ली गई है।"); say( "hotelMapMsg", "Hotel चुना गया। Pin को खिसकाकर सही जगह भी चुन सकते हैं।" ); } function searchHotels(){ const query=$("hotelName").value.trim(); clearTimeout(hotelSearchTimer); if(hotelSearchController){ hotelSearchController.abort(); } if(query.length<3){ $("hotelSearchResults").replaceChildren(); say("hotelSearchMsg","कम से कम 3 अक्षर लिखें।"); return; } // पुरानी Hotel Location हटाएँ selectedHotelLocation=false; $("hotelAddress").value=""; $("hotelLat").value=""; $("hotelLng").value=""; $("hotelGoogleMapsLink").classList.add("hidden"); $("hotelGoogleMapsLink").removeAttribute("href"); if(hotelMarker&&hotelMap){ hotelMap.removeLayer(hotelMarker); hotelMarker=null; } hotelSearchTimer=setTimeout(async()=>{ const host=$("hotelSearchResults"); host.replaceChildren(); say("hotelSearchMsg","OpenStreetMap पर होटल खोजे जा रहे हैं..."); hotelSearchController=new AbortController(); try{ const params=new URLSearchParams(); // केवल होटल का नाम खोजें params.set("q",query); params.set("limit","10"); params.set("lang","en"); // Bhilwara के आसपास की खोज को प्राथमिकता const city=$("hotelCity").value.trim().toLowerCase(); if(!city||city.includes("bhilwara")){ params.set("lat","25.34"); params.set("lon","74.63"); } const url="https://photon.komoot.io/api/?"+params.toString(); const response=await fetch(url,{ signal:hotelSearchController.signal }); if(!response.ok){ throw new Error("Search Service अभी उपलब्ध नहीं है।"); } const result=await response.json(); const features=result.features||[]; host.replaceChildren(); if(!features.length){ say("hotelSearchMsg","होटल नहीं मिला। पूरा नाम या दूसरा नाम लिखें।"); return; } let count=0; features.forEach(feature=>{ const p=feature.properties||{}; const coords=feature.geometry?.coordinates; if(!coords||coords.length<2)return; const title=p.name||"Unnamed Place"; const address=makeHotelAddress(p); const button=document.createElement("button"); button.className="hotel-result"; button.textContent= "🏨 "+title+"\n"+ (address||"पूरा पता उपलब्ध नहीं")+"\n"+ "📍 Map Location उपलब्ध"; button.onclick=()=>selectHotelResult(feature); host.appendChild(button); count++; }); say( "hotelSearchMsg", count? count+" सुझाव मिले। सही होटल चुनें।": "कोई सही GPS Location नहीं मिली।" ); }catch(e){ if(e.name==="AbortError")return; host.replaceChildren(); say("hotelSearchMsg","Search Error: "+e.message); } },700); } $("hotelName").addEventListener("input",searchHotels); $("clearHotelLocation").onclick=clearHotelLocation; // Map initialize initHotelMap();
